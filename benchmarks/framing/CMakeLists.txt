if(NOT TARGET benchmark::benchmark)
  set(WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR "" CACHE PATH "Optional Google Benchmark source checkout")
  if(WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR}"
      "${CMAKE_CURRENT_BINARY_DIR}/google-benchmark" EXCLUDE_FROM_ALL)
  else()
    find_package(benchmark 1.9 CONFIG REQUIRED)
  endif()
endif()
add_executable(wirelink_framing_benchmark micro.cpp workload.c)
target_link_libraries(wirelink_framing_benchmark PRIVATE Wirelink::wirelink benchmark::benchmark)
target_compile_features(wirelink_framing_benchmark PRIVATE cxx_std_17)
set(_wirelink_framing_fixed_default OFF)
if(WIRELINK_LINK_PROFILE STREQUAL "cobs-crc16")
  set(_wirelink_framing_fixed_default ON)
endif()
option(WIRELINK_FRAMING_COBS_CRC16_ONLY "Limit the diagnostic workload matrix to COBS + CRC16" ${_wirelink_framing_fixed_default})
if(WIRELINK_FRAMING_COBS_CRC16_ONLY)
  target_compile_definitions(wirelink_framing_benchmark PRIVATE WL_FRAMING_COBS_CRC16_ONLY=1)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(wirelink_framing_benchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_test(NAME wirelink_framing_benchmark COMMAND wirelink_framing_benchmark
    --benchmark_min_time=0.001s --benchmark_repetitions=1)
  set_tests_properties(wirelink_framing_benchmark PROPERTIES TIMEOUT 120 LABELS "benchmark;smoke")
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME wirelink_framing_report_tests
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_compare.py")
    set_tests_properties(wirelink_framing_report_tests PROPERTIES TIMEOUT 30)
  endif()
endif()
