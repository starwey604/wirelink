cmake_minimum_required(VERSION 3.28.0)
set(RAGTIME_FIRMWARES_ROOT "" CACHE PATH "Existing Ragtime Zephyr workspace (read-only source)")
if(NOT EXISTS "${RAGTIME_FIRMWARES_ROOT}/firmware/tests/apps/willow_wirelink_hil/src/main.cpp")
  message(FATAL_ERROR "Set RAGTIME_FIRMWARES_ROOT to the existing product workspace")
endif()
set(ENV{RAGTIME_FIRMWARES_ROOT} "${RAGTIME_FIRMWARES_ROOT}")
set(HIL_ROOT "${RAGTIME_FIRMWARES_ROOT}/firmware/tests/apps/willow_wirelink_hil")
set(CONF_FILE "${HIL_ROOT}/prj.conf;${HIL_ROOT}/boards/dm_mc02_stm32h723xx.conf;${CMAKE_CURRENT_SOURCE_DIR}/prj.conf")
set(DTC_OVERLAY_FILE "${HIL_ROOT}/boards/dm_mc02_stm32h723xx.overlay")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(wirelink_willow_cpu LANGUAGES C CXX)
option(WIRELINK_H7_LOCK_PROFILING "Measure ArmProtocol state spinlocks (test-only)" ON)
target_sources(app PRIVATE "${HIL_ROOT}/src/main.cpp" src/main.cpp src/spin_profile.cpp)
set_source_files_properties("${HIL_ROOT}/src/main.cpp" PROPERTIES COMPILE_DEFINITIONS "main=willow_hil_main")
target_link_libraries(app PRIVATE fci_protocol::arm)
if(WIRELINK_H7_LOCK_PROFILING)
  get_property(libraries GLOBAL PROPERTY ZEPHYR_LIBS)
  set(found FALSE)
  foreach(library IN LISTS libraries)
    if(library MATCHES "arm_protocol$")
      # Force include ONLY in this test's ArmProtocol library, not kernel/USB.
      target_compile_options(${library} PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/src/spin_profile.hpp")
      set(found TRUE)
    endif()
  endforeach()
  if(NOT found)
    message(FATAL_ERROR "ArmProtocol target was not found; refusing an uninstrumented result")
  endif()
  target_compile_definitions(app PRIVATE WL_H7_LOCK_PROFILING=1)
endif()
