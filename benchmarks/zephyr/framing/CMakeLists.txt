cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(wirelink_framing_cpu LANGUAGES C)
target_sources(app PRIVATE src/main.c ../../framing/workload.c)
target_include_directories(app PRIVATE ../../framing)
set(_wirelink_framing_fixed_default OFF)
if(CONFIG_WIRELINK_LINK_PROFILE_COBS_CRC16)
  set(_wirelink_framing_fixed_default ON)
endif()
option(WIRELINK_FRAMING_COBS_CRC16_ONLY "Limit the diagnostic workload matrix to COBS + CRC16" ${_wirelink_framing_fixed_default})
if(WIRELINK_FRAMING_COBS_CRC16_ONLY)
  target_compile_definitions(app PRIVATE WL_FRAMING_COBS_CRC16_ONLY=1)
endif()
