cmake_minimum_required(VERSION 3.20.0)
set(WIRELINK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
list(APPEND ZEPHYR_EXTRA_MODULES ${WIRELINK_ROOT})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(wirelink_framing_cpu LANGUAGES C)
target_sources(app PRIVATE src/main.c ../../framing/workload.c)
target_include_directories(app PRIVATE ../../framing)
option(WIRELINK_FRAMING_COBS_CRC16_ONLY "Limit the independent tool matrix to COBS + CRC16" OFF)
if(WIRELINK_FRAMING_COBS_CRC16_ONLY)
  target_compile_definitions(app PRIVATE WL_FRAMING_COBS_CRC16_ONLY=1)
endif()
