cmake_minimum_required(VERSION 3.21)
set(HARNESS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(WIRELINK_SOURCE_DIR "${HARNESS_ROOT}" CACHE PATH "Core under test; may be a frozen Git export")
list(APPEND ZEPHYR_EXTRA_MODULES "${WIRELINK_SOURCE_DIR}")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(wirelink_mixed_traffic_h7 LANGUAGES C)
option(MIXED_EXPECT_COEXIST "Expect unreliable progress while reliable ACK is pending" ON)
if(NOT CONFIG_LTO)
  message(FATAL_ERROR "This paired H7 measurement requires LTO")
endif()
list(APPEND CMAKE_MODULE_PATH "${HARNESS_ROOT}/cmake")
include(WirelinkWlc)
add_library(Wirelink::wirelink ALIAS wirelink)
set(WORKLOAD "${HARNESS_ROOT}/benchmarks/mixed_traffic")
wirelink_wlc_generate_codec(TARGET mixed_codec SCHEMA "${WORKLOAD}/mixed.wl")
target_link_libraries(mixed_codec PUBLIC zephyr_interface)
wirelink_wlc_generate_runtime(TARGET mixed_runtime CODEC_TARGET mixed_codec
  PROFILE "${WORKLOAD}/mixed.bind.wl" RUNTIME_NAME mixed)
target_sources(app PRIVATE src/main.c src/retry_cpu.c "${WORKLOAD}/workload.c")
target_include_directories(app PRIVATE "${WORKLOAD}")
target_compile_definitions(app PRIVATE MIXED_TRAFFIC_H7=1
  MIXED_EXPECT_COEXIST=$<BOOL:${MIXED_EXPECT_COEXIST}>)
target_link_libraries(app PRIVATE mixed_runtime)
target_compile_features(app PRIVATE c_std_11)
