cmake_minimum_required(VERSION 3.21)
set(HARNESS_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
set(WIRELINK_SOURCE_DIR "${HARNESS_ROOT}" CACHE PATH "Core under test; may be a frozen Git export")
list(APPEND ZEPHYR_EXTRA_MODULES "${WIRELINK_SOURCE_DIR}")
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(wirelink_mixed_traffic LANGUAGES C)
if(DEFINED MIXED_EXPECT_COEXIST)
  message(FATAL_ERROR "MIXED_EXPECT_COEXIST was removed with ARQ; use a fresh build directory")
endif()
include(${HARNESS_ROOT}/cmake/WirelinkWlc.cmake)
set(WORKLOAD "${HARNESS_ROOT}/benchmarks/mixed_traffic")
wirelink_wlc_generate(TARGET mixed_protocol SCHEMA "${WORKLOAD}/mixed.wl"
  PROFILE "${WORKLOAD}/mixed.bind.wl")
target_sources(app PRIVATE src/main.c "${WORKLOAD}/workload.c")
target_include_directories(app PRIVATE "${WORKLOAD}")
target_compile_definitions(app PRIVATE MIXED_TRAFFIC_FIRMWARE=1)
target_link_libraries(app PRIVATE mixed_protocol)
target_compile_features(app PRIVATE c_std_11)
