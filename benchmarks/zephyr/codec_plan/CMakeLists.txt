cmake_minimum_required(VERSION 3.20)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(wirelink_codec_plan_cpu LANGUAGES C)
if(NOT DEFINED CODEC_PLAN_EXPECT_LTO)
  set(CODEC_PLAN_EXPECT_LTO ON)
endif()
if(CODEC_PLAN_EXPECT_LTO AND NOT CONFIG_LTO)
  message(FATAL_ERROR "LTO requested but not enabled by Kconfig; do not label this an LTO measurement")
elseif(NOT CODEC_PLAN_EXPECT_LTO AND CONFIG_LTO)
  message(FATAL_ERROR "No-LTO comparison unexpectedly enabled LTO")
endif()
if(NOT EXISTS "${CODEC_PLAN_CODEC_DIR}/plan.c")
  message(FATAL_ERROR "Provide frozen generated CODEC_PLAN_CODEC_DIR")
endif()
target_sources(app PRIVATE src/main.c ../../codec_plan/workload.c "${CODEC_PLAN_CODEC_DIR}/plan.c")
target_include_directories(app PRIVATE ../../codec_plan "${CODEC_PLAN_CODEC_DIR}")
