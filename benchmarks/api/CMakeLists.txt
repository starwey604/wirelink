find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(NOT TARGET benchmark::benchmark)
  set(WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR "" CACHE PATH "Optional Google Benchmark checkout")
  if(WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR}" google-benchmark EXCLUDE_FROM_ALL)
  else()
    find_package(benchmark 1.9 CONFIG REQUIRED)
  endif()
endif()
wirelink_wlc_generate(TARGET api_protocol SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/kernel_api.wl"
  PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/kernel_api.bind.wl")
add_executable(wirelink_api_benchmark micro.cpp fixture.c)
target_compile_features(wirelink_api_benchmark PRIVATE cxx_std_20)
target_include_directories(wirelink_api_benchmark PRIVATE ../../examples/common)
target_link_libraries(wirelink_api_benchmark PRIVATE api_protocol Wirelink::packet_loopback benchmark::benchmark)
wl_warnings(wirelink_api_benchmark PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
  add_test(NAME api_kernel_smoke COMMAND wirelink_api_benchmark --benchmark_min_time=0.001s --benchmark_repetitions=1)
  set_tests_properties(api_kernel_smoke PROPERTIES TIMEOUT 30 LABELS "benchmark;smoke")
  add_test(NAME api_current_reports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_kernel_reports.py)
  # Historical readers stay tested against archived workload fixtures.
  add_test(NAME api_historical_reports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_reports.py)
  add_test(NAME executor_historical_reports COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_executor_reports.py)
endif()
