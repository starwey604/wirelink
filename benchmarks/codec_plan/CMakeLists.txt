cmake_minimum_required(VERSION 3.21)
project(wirelink_codec_plan LANGUAGES C CXX)
include(CTest)
if(NOT EXISTS "${CODEC_PLAN_CODEC_DIR}/plan.c")
  message(FATAL_ERROR "Generate plan.wl separately and set CODEC_PLAN_CODEC_DIR")
endif()
if(WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BENCHMARK_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${WIRELINK_GOOGLE_BENCHMARK_SOURCE_DIR}" benchmark EXCLUDE_FROM_ALL)
else()
  find_package(benchmark 1.9 CONFIG REQUIRED)
endif()
add_library(codec_plan_workload STATIC workload.c "${CODEC_PLAN_CODEC_DIR}/plan.c")
target_include_directories(codec_plan_workload PUBLIC "${CODEC_PLAN_CODEC_DIR}" . ../../include)
target_compile_features(codec_plan_workload PUBLIC c_std_11)
add_executable(wirelink_codec_plan_benchmark micro.cpp)
target_link_libraries(wirelink_codec_plan_benchmark PRIVATE codec_plan_workload benchmark::benchmark)
target_compile_features(wirelink_codec_plan_benchmark PRIVATE cxx_std_17)
add_test(NAME codec_plan_smoke COMMAND wirelink_codec_plan_benchmark --benchmark_min_time=0.001s)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME codec_plan_compare_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_compare.py")
endif()
