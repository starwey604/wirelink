cmake_minimum_required(VERSION 3.21)
project(wirelink_udp_package_consumer LANGUAGES C CXX)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Wirelink 0.13.0 EXACT CONFIG REQUIRED COMPONENTS asio_udp)
add_executable(udp_consumer main.cpp owner.c)
target_compile_features(udp_consumer PRIVATE cxx_std_20)
target_link_libraries(udp_consumer PRIVATE Wirelink::asio_udp)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(udp_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME wirelink_udp_package_consumer COMMAND udp_consumer)
set_tests_properties(wirelink_udp_package_consumer PROPERTIES TIMEOUT 10)
