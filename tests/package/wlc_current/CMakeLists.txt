cmake_minimum_required(VERSION 3.21)
project(wirelink_current_generator_consumer LANGUAGES C CXX)
include(CTest)
find_package(Wirelink CONFIG REQUIRED)
get_filename_component(fixture "${CMAKE_CURRENT_SOURCE_DIR}/../static_endpoint" ABSOLUTE)
file(READ "${fixture}/probe.wl" schema)
string(REPLACE "io = external;" "io = embedded; framing = cobs_crc16;" schema "${schema}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/cobs-schema")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/cobs-schema/probe.wl" "${schema}")
wirelink_wlc_generate(TARGET probe_native SCHEMA "${fixture}/probe.wl")
wirelink_wlc_generate(TARGET probe_cobs SCHEMA "${CMAKE_CURRENT_BINARY_DIR}/cobs-schema/probe.wl")
add_executable(installed_c "${fixture}/native.c")
target_link_libraries(installed_c PRIVATE probe_native Wirelink::packet_loopback)
add_executable(installed_cpp "${fixture}/owner.c" "${fixture}/consumer.cpp")
target_compile_features(installed_cpp PRIVATE cxx_std_20)
target_link_libraries(installed_cpp PRIVATE probe_native)
add_executable(installed_cobs "${fixture}/cobs.c")
target_link_libraries(installed_cobs PRIVATE probe_cobs)
foreach(target installed_c installed_cpp installed_cobs)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
# Keyword escaping is shared with WLC; no CMake-side copy of identifier rules.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/switch.wl" "version 1; message Empty @id(1) {}\n")
wirelink_wlc_generate_codec(TARGET keyword_codec SCHEMA "${CMAKE_CURRENT_BINARY_DIR}/switch.wl")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/keyword.c"
  "#include \"switch_.h\"\nint main(void) { empty_t value; empty_clear(&value); return 0; }\n")
add_executable(installed_keyword "${CMAKE_CURRENT_BINARY_DIR}/keyword.c")
target_link_libraries(installed_keyword PRIVATE keyword_codec)
add_test(NAME installed_keyword COMMAND installed_keyword)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME generator_diagnostics COMMAND ${Python3_EXECUTABLE}
  "${CMAKE_CURRENT_SOURCE_DIR}/check_diagnostics.py" --cmake "${CMAKE_COMMAND}"
  --prefix "${CMAKE_PREFIX_PATH}" --wlc "${WIRELINK_WLC_EXECUTABLE}")
# Compile/link with the consumer defaults: no function-section/GC requirement.
wirelink_wlc_generate(TARGET roles_client SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/roles.wl" NODE client)
wirelink_wlc_generate(TARGET roles_all SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/roles.wl" ALL_NODES)
add_executable(installed_client_only client_only.c)
target_link_libraries(installed_client_only PRIVATE roles_client)
add_executable(installed_all_nodes all_nodes.c)
target_link_libraries(installed_all_nodes PRIVATE roles_all)
foreach(target installed_client_only installed_all_nodes)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
