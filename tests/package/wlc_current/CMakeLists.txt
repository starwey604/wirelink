cmake_minimum_required(VERSION 3.21)
project(wirelink_current_generator_consumer LANGUAGES C CXX)
include(CTest)
find_package(Wirelink CONFIG REQUIRED)
get_filename_component(fixture "${CMAKE_CURRENT_SOURCE_DIR}/../static_kernel" ABSOLUTE)
file(READ "${fixture}/probe.bind.wl" profile)
string(REPLACE "native_packet" "cobs_stream" profile "${profile}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe.cobs.bind.wl" "${profile}")
wirelink_wlc_generate(TARGET probe_native SCHEMA "${fixture}/probe.wl" PROFILE "${fixture}/probe.bind.wl")
wirelink_wlc_generate(TARGET probe_cobs SCHEMA "${fixture}/probe.wl" PROFILE "${CMAKE_CURRENT_BINARY_DIR}/probe.cobs.bind.wl")
add_executable(installed_c "${fixture}/native.c")
target_link_libraries(installed_c PRIVATE probe_native Wirelink::packet_loopback)
add_executable(installed_cpp "${fixture}/owner.c" "${fixture}/consumer.cpp")
target_compile_features(installed_cpp PRIVATE cxx_std_20)
target_link_libraries(installed_cpp PRIVATE probe_native)
add_executable(installed_cobs "${fixture}/cobs.c")
target_link_libraries(installed_cobs PRIVATE probe_cobs)
foreach(target installed_c installed_cpp installed_cobs)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endforeach()
# Keyword escaping is shared with WLC; no CMake-side copy of identifier rules.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/switch.wl" "version 1; message Empty @id(1) {}\n")
wirelink_wlc_generate_codec(TARGET keyword_codec SCHEMA "${CMAKE_CURRENT_BINARY_DIR}/switch.wl")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/keyword.c"
  "#include \"switch_.h\"\nint main(void) { empty_value_t value; empty_value_clear(&value); return 0; }\n")
add_executable(installed_keyword "${CMAKE_CURRENT_BINARY_DIR}/keyword.c")
target_link_libraries(installed_keyword PRIVATE keyword_codec)
add_test(NAME installed_keyword COMMAND installed_keyword)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME generator_diagnostics COMMAND ${Python3_EXECUTABLE}
  "${CMAKE_CURRENT_SOURCE_DIR}/check_diagnostics.py" --cmake "${CMAKE_COMMAND}"
  --prefix "${CMAKE_PREFIX_PATH}" --wlc "${WIRELINK_WLC_EXECUTABLE}")
