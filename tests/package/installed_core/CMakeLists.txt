cmake_minimum_required(VERSION 3.21)
project(wirelink_installed_consumer LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(WIRELINK_CONSUMER_C_ONLY "Test core CMake and pkg-config consumers without C++" OFF)
option(WIRELINK_CONSUMER_HOST "Test installed Executor, UDP and SDK support" OFF)
set(WIRELINK_TEST_ASTRIAL_SOURCE_DIR "" CACHE PATH "Source for the installed Astrial backends")

find_package(Wirelink CONFIG REQUIRED)
if(CMAKE_CXX_COMPILER_LOADED)
  message(FATAL_ERROR "Finding the core package enabled C++")
endif()

function(consumer target source)
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE ${ARGN})
  set_target_properties(${target} PROPERTIES NO_SYSTEM_FROM_IMPORTED YES)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  endif()
  add_test(NAME ${target} COMMAND ${target})
endfunction()

consumer(installed_public_c public.c Wirelink::wirelink)
# The exported usage requirements upgrade consumers with an older C baseline.
set_target_properties(installed_public_c PROPERTIES C_STANDARD 90)
if(WIRELINK_CONSUMER_C_ONLY)
  get_target_property(includes Wirelink::packet_port INTERFACE_INCLUDE_DIRECTORIES)
  foreach(directory IN LISTS includes)
    foreach(retired link.h kernel.h pump.h rpc.h rpc_async.h rpc_sync.h)
      if(EXISTS "${directory}/wirelink/${retired}")
        message(FATAL_ERROR "Installed legacy API: ${retired}")
      endif()
    endforeach()
  endforeach()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(WIRELINK_PC REQUIRED IMPORTED_TARGET wirelink)
  consumer(installed_core_cmake pkgconfig.c Wirelink::wirelink)
  consumer(installed_core_pkgconfig pkgconfig.c PkgConfig::WIRELINK_PC)
  return()
endif()

enable_language(CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
consumer(installed_public_cpp public.cpp Wirelink::wirelink)
if(WIRELINK_CONSUMER_HOST)
  find_package(Wirelink CONFIG REQUIRED COMPONENTS cpp_host udp)
  consumer(installed_host host.cpp Wirelink::host)
  consumer(installed_udp_c udp.c Wirelink::udp)
  consumer(installed_udp_cpp udp.cpp Wirelink::udp)
  set_tests_properties(installed_host installed_udp_c installed_udp_cpp PROPERTIES TIMEOUT 10)
endif()
if(WIRELINK_WLC_EXECUTABLE)
  include(generator.cmake)
endif()
if(WIRELINK_TEST_ASTRIAL_SOURCE_DIR)
  include(astrial.cmake)
endif()
