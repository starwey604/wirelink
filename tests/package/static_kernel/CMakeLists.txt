cmake_minimum_required(VERSION 3.21)
project(wirelink_kernel_installed_consumer LANGUAGES C CXX)
include(CTest)
find_package(WirelinkKernel CONFIG REQUIRED)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(WIRELINK_KERNEL_WLC_EXECUTABLE "" CACHE FILEPATH "Matching WLC")
if(NOT EXISTS "${WIRELINK_KERNEL_WLC_EXECUTABLE}")
  message(FATAL_ERROR "Set WIRELINK_KERNEL_WLC_EXECUTABLE to matching WLC with compile-kernel")
endif()
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/probe.bind.wl profile)
string(REPLACE "native_packet" "cobs_stream" profile "${profile}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/probe.cobs.bind.wl "${profile}")
foreach(mode native cobs)
  set(generated ${CMAKE_CURRENT_BINARY_DIR}/${mode})
  set(binding ${CMAKE_CURRENT_SOURCE_DIR}/probe.bind.wl)
  if(mode STREQUAL cobs)
    set(binding ${CMAKE_CURRENT_BINARY_DIR}/probe.cobs.bind.wl)
    set(transport ${generated}/probe_kernel_transport.c)
  else()
    set(transport)
  endif()
  add_custom_command(OUTPUT ${generated}/probe.c ${generated}/probe_kernel.c
      ${generated}/probe.h ${generated}/probe_values.h ${generated}/probe_kernel.h
      ${generated}/probe_kernel_codecs.h ${transport}
    COMMAND ${WIRELINK_KERNEL_WLC_EXECUTABLE} compile-kernel ${CMAKE_CURRENT_SOURCE_DIR}/probe.wl
      --profile ${binding} --out-dir ${generated}
    DEPENDS ${WIRELINK_KERNEL_WLC_EXECUTABLE} probe.wl ${binding} VERBATIM)
  add_library(probe_${mode} ${generated}/probe.c ${generated}/probe_kernel.c ${transport})
  target_include_directories(probe_${mode} PUBLIC ${generated})
  target_link_libraries(probe_${mode} PUBLIC WirelinkKernel::kernel)
  target_compile_options(probe_${mode} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
target_link_libraries(probe_cobs PUBLIC WirelinkKernel::stream_cobs_crc16)
add_executable(installed_c native.c)
target_link_libraries(installed_c PRIVATE probe_native WirelinkKernel::packet_loopback)
add_executable(installed_cpp owner.c consumer.cpp)
target_compile_features(installed_cpp PRIVATE cxx_std_20)
target_link_libraries(installed_cpp PRIVATE probe_native)
add_executable(installed_cobs cobs.c)
target_link_libraries(installed_cobs PRIVATE probe_cobs)
foreach(target installed_c installed_cpp installed_cobs)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
