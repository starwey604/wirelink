cmake_minimum_required(VERSION 3.21)
project(wirelink_c_only_installed LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Wirelink CONFIG REQUIRED)
if(CMAKE_CXX_COMPILER_LOADED)
  message(FATAL_ERROR "The core package must not enable C++")
endif()
get_target_property(_include Wirelink::packet_port INTERFACE_INCLUDE_DIRECTORIES)
foreach(_dir IN LISTS _include)
  foreach(_retired link.h endpoint.h pump.h rpc.h rpc_async.h rpc_sync.h)
    if(EXISTS "${_dir}/wirelink/${_retired}")
      message(FATAL_ERROR "Installed legacy API: ${_retired}")
    endif()
  endforeach()
endforeach()
find_package(PkgConfig REQUIRED)
pkg_check_modules(WIRELINK_PC REQUIRED IMPORTED_TARGET wirelink)
foreach(_mode cmake pkgconfig)
  add_executable(installed_core_${_mode} consumer.c)
  if(_mode STREQUAL cmake)
    target_link_libraries(installed_core_${_mode} PRIVATE Wirelink::wirelink)
  else()
    target_link_libraries(installed_core_${_mode} PRIVATE PkgConfig::WIRELINK_PC)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(installed_core_${_mode} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME installed_core_${_mode} COMMAND installed_core_${_mode})
endforeach()
