cmake_minimum_required(VERSION 3.21)
project(wirelink_installed_adapters LANGUAGES C CXX)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(WIRELINK_TEST_ASTRIAL_SOURCE_DIR "" CACHE PATH "Application-supplied unchanged Astrial backend")
if(NOT EXISTS "${WIRELINK_TEST_ASTRIAL_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set WIRELINK_TEST_ASTRIAL_SOURCE_DIR")
endif()
set(ASTRIAL_BUILD_USB ON CACHE BOOL "" FORCE)
set(ASTRIAL_ENABLE_IPO OFF CACHE BOOL "" FORCE)
add_subdirectory(${WIRELINK_TEST_ASTRIAL_SOURCE_DIR} astrial EXCLUDE_FROM_ALL)
find_package(Wirelink CONFIG REQUIRED COMPONENTS asio_udp astrial astrial_usb)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(_case udp usb serial)
  if(_case STREQUAL udp)
    set(_component asio_udp)
  elseif(_case STREQUAL usb)
    set(_component astrial_usb)
  else()
    if(NOT UNIX)
      continue()
    endif()
    set(_component astrial)
  endif()
  add_executable(installed_${_case} ${_root}/tests/core/packet_${_case}.cpp
    ${_root}/tests/core/packet_${_case}_owner.c)
  target_compile_features(installed_${_case} PRIVATE cxx_std_20)
  target_link_libraries(installed_${_case} PRIVATE Wirelink::${_component})
  if(_case STREQUAL udp)
    target_link_libraries(installed_${_case} PRIVATE Wirelink::stream_cobs_crc16)
  elseif(_case STREQUAL serial AND CMAKE_SYSTEM_NAME STREQUAL Linux)
    target_link_libraries(installed_${_case} PRIVATE util)
  endif()
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(installed_${_case} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME installed_${_case} COMMAND installed_${_case})
  set_tests_properties(installed_${_case} PROPERTIES TIMEOUT 10)
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME installed_component_diagnostics COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/check_components.py ${CMAKE_COMMAND} "${CMAKE_PREFIX_PATH}"
  ${CMAKE_CURRENT_SOURCE_DIR}/../missing_backend ${CMAKE_CURRENT_BINARY_DIR}/negative)
