add_library(wirelink_diagnostics src/diagnostics.c)
add_library(Wirelink::diagnostics ALIAS wirelink_diagnostics)
set_target_properties(wirelink_diagnostics PROPERTIES EXPORT_NAME diagnostics)
target_link_libraries(wirelink_diagnostics PUBLIC wirelink_types)
wl_warnings(wirelink_diagnostics PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
  add_executable(wirelink_diagnostics_test tests/diagnostics.c)
  target_link_libraries(wirelink_diagnostics_test PRIVATE wirelink_diagnostics)
  wl_warnings(wirelink_diagnostics_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME diagnostics COMMAND wirelink_diagnostics_test)
endif()
if(WIRELINK_INSTALL)
  install(TARGETS wirelink_diagnostics EXPORT WirelinkTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
