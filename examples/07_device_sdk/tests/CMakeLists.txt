add_library(device_test_service STATIC service.c server_bridge.c)
set_target_properties(device_test_service PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(device_test_service PUBLIC . ../generated ../src)
target_link_libraries(device_test_service PUBLIC DeviceSdk::client Wirelink::asio_udp)
add_executable(device_cpp_tests client.cpp)
target_link_libraries(device_cpp_tests PRIVATE device_test_service)
add_test(NAME device_cpp COMMAND device_cpp_tests)
set_tests_properties(device_cpp PROPERTIES TIMEOUT 30)
add_executable(device_c_server server_main.cpp)
target_link_libraries(device_c_server PRIVATE device_test_service)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target device_test_service device_cpp_tests device_c_server)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()
if(MSVC)
  target_compile_options(device_cpp_tests PRIVATE /utf-8)
endif()
