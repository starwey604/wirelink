cmake_minimum_required(VERSION 3.21)
project(wirelink_static_kernel_prototype LANGUAGES C CXX)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(wirelink_packet_port src/packet_port.c)
target_include_directories(wirelink_packet_port PUBLIC include ../../include)
add_library(wirelink_kernel_wire src/native_none.c src/rpc_wire.c)
target_include_directories(wirelink_kernel_wire PUBLIC include ../../include)
add_library(wirelink_kernel src/kernel.c)
target_link_libraries(wirelink_kernel PUBLIC wirelink_packet_port wirelink_kernel_wire)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(wirelink_packet_port PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(wirelink_kernel_wire PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(wirelink_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(packet_port_test tests/packet_port.c)
  target_link_libraries(packet_port_test PRIVATE wirelink_packet_port)
  add_test(NAME packet_port COMMAND packet_port_test)
  find_package(Threads REQUIRED)
  add_executable(packet_port_concurrent_test tests/packet_port_concurrent.c)
  target_link_libraries(packet_port_concurrent_test PRIVATE wirelink_packet_port Threads::Threads)
  add_test(NAME packet_port_concurrent COMMAND packet_port_concurrent_test)
  add_executable(kernel_wire_test tests/kernel_wire.c)
  target_link_libraries(kernel_wire_test PRIVATE wirelink_kernel_wire)
  add_test(NAME kernel_wire COMMAND kernel_wire_test)
  add_executable(kernel_test tests/kernel.c)
  target_link_libraries(kernel_test PRIVATE wirelink_kernel)
  add_test(NAME kernel COMMAND kernel_test)
  set(WIRELINK_KERNEL_WLC_EXECUTABLE "" CACHE FILEPATH "WLC with compile-kernel support")
  if(WIRELINK_KERNEL_WLC_EXECUTABLE)
    set(_generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
    add_custom_command(OUTPUT ${_generated}/usb.c ${_generated}/usb.h
        ${_generated}/usb_values.h ${_generated}/usb_kernel.c ${_generated}/usb_kernel.h
        ${_generated}/usb_kernel_codecs.h
        ${_generated}/usb_kernel_manifest.json ${_generated}/usb_kernel_deployment.json
      COMMAND ${WIRELINK_KERNEL_WLC_EXECUTABLE} compile-kernel
        ${CMAKE_CURRENT_SOURCE_DIR}/schema/usb.wl
        --profile ${CMAKE_CURRENT_SOURCE_DIR}/schema/usb.bind.wl --out-dir ${_generated}
      DEPENDS ${WIRELINK_KERNEL_WLC_EXECUTABLE} schema/usb.wl schema/usb.bind.wl VERBATIM)
    add_library(kernel_usb_fixture_codec ${_generated}/usb.c ${_generated}/usb_kernel.c)
    target_include_directories(kernel_usb_fixture_codec PUBLIC ${_generated})
    target_link_libraries(kernel_usb_fixture_codec PUBLIC wirelink_kernel)
    target_compile_options(kernel_usb_fixture_codec PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_executable(generated_usb_test tests/generated_usb.c)
    target_link_libraries(generated_usb_test PRIVATE kernel_usb_fixture_codec)
    add_test(NAME generated_usb COMMAND generated_usb_test)
    add_executable(kernel_cpp_test tests/cpp_owner.c tests/cpp_consumer.cpp)
    target_compile_features(kernel_cpp_test PRIVATE cxx_std_20)
    target_compile_options(kernel_cpp_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(kernel_cpp_test PRIVATE kernel_usb_fixture_codec)
    add_test(NAME kernel_cpp COMMAND kernel_cpp_test)
    set(_instrument "${CMAKE_CURRENT_BINARY_DIR}/instrument")
    add_custom_command(OUTPUT ${_instrument}/instrument.c ${_instrument}/instrument.h
        ${_instrument}/instrument_values.h ${_instrument}/instrument_kernel.c
        ${_instrument}/instrument_kernel.h ${_instrument}/instrument_kernel_codecs.h
        ${_instrument}/instrument_kernel_manifest.json ${_instrument}/instrument_kernel_deployment.json
      COMMAND ${WIRELINK_KERNEL_WLC_EXECUTABLE} compile-kernel
        ${CMAKE_CURRENT_SOURCE_DIR}/schema/instrument.wl
        --profile ${CMAKE_CURRENT_SOURCE_DIR}/schema/instrument.bind.wl --out-dir ${_instrument}
      DEPENDS ${WIRELINK_KERNEL_WLC_EXECUTABLE} schema/instrument.wl schema/instrument.bind.wl VERBATIM)
    add_executable(generated_instrument_test tests/generated_instrument.c
      ${_instrument}/instrument.c ${_instrument}/instrument_kernel.c)
    target_include_directories(generated_instrument_test PRIVATE ${_instrument})
    target_compile_options(generated_instrument_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(generated_instrument_test PRIVATE wirelink_kernel)
    add_test(NAME generated_instrument COMMAND generated_instrument_test)
  else()
    message(STATUS "Generated endpoint tests need WIRELINK_KERNEL_WLC_EXECUTABLE")
  endif()
endif()

option(WIRELINK_KERNEL_BUILD_ASTRIAL_USB "Build isolated packet USB host adapter" OFF)
if(WIRELINK_KERNEL_BUILD_ASTRIAL_USB)
  if(NOT TARGET astrial_usb)
    set(WIRELINK_KERNEL_ASTRIAL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../astrial"
      CACHE PATH "Astrial source")
    set(ASTRIAL_BUILD_USB ON CACHE BOOL "" FORCE)
    set(ASTRIAL_ENABLE_IPO OFF CACHE BOOL "" FORCE)
    add_subdirectory(${WIRELINK_KERNEL_ASTRIAL_SOURCE_DIR} astrial EXCLUDE_FROM_ALL)
  endif()
  add_library(wirelink_kernel_astrial_usb adapters/astrial/packet_usb.cpp)
  target_compile_features(wirelink_kernel_astrial_usb PUBLIC cxx_std_20)
  target_include_directories(wirelink_kernel_astrial_usb PUBLIC adapters/astrial/include ../../include)
  target_link_libraries(wirelink_kernel_astrial_usb PUBLIC wirelink_packet_port astrial_usb)
  target_compile_options(wirelink_kernel_astrial_usb PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(BUILD_TESTING)
    add_executable(kernel_packet_usb_test tests/packet_usb_owner.c tests/packet_usb.cpp)
    target_compile_options(kernel_packet_usb_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(kernel_packet_usb_test PRIVATE wirelink_kernel_astrial_usb)
    add_test(NAME kernel_packet_usb COMMAND kernel_packet_usb_test)
    set_tests_properties(kernel_packet_usb PROPERTIES TIMEOUT 10)
  endif()
endif()

option(WIRELINK_KERNEL_BUILD_ASIO_UDP "Build isolated packet UDP adapter" OFF)
if(WIRELINK_KERNEL_BUILD_ASIO_UDP)
  find_path(WIRELINK_KERNEL_ASIO_INCLUDE_DIR asio.hpp
    HINTS ${CMAKE_CURRENT_SOURCE_DIR}/../../../astrial/3rdparty/asio/include)
  if(NOT WIRELINK_KERNEL_ASIO_INCLUDE_DIR)
    message(FATAL_ERROR "Set WIRELINK_KERNEL_ASIO_INCLUDE_DIR to standalone Asio headers")
  endif()
  find_package(Threads REQUIRED)
  add_library(wirelink_kernel_asio_udp adapters/asio/packet_udp.cpp)
  target_compile_features(wirelink_kernel_asio_udp PUBLIC cxx_std_20)
  target_compile_definitions(wirelink_kernel_asio_udp PRIVATE ASIO_STANDALONE)
  target_include_directories(wirelink_kernel_asio_udp PUBLIC adapters/asio/include
    PRIVATE ${WIRELINK_KERNEL_ASIO_INCLUDE_DIR})
  target_link_libraries(wirelink_kernel_asio_udp PUBLIC wirelink_packet_port PRIVATE wirelink_kernel_wire Threads::Threads)
  target_compile_options(wirelink_kernel_asio_udp PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(WIN32)
    target_compile_definitions(wirelink_kernel_asio_udp PRIVATE _WIN32_WINNT=0x0601 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(wirelink_kernel_asio_udp PRIVATE ws2_32 mswsock)
  endif()
  if(BUILD_TESTING)
    add_executable(kernel_packet_udp_test tests/packet_udp_owner.c tests/packet_udp.cpp)
    target_compile_options(kernel_packet_udp_test PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(kernel_packet_udp_test PRIVATE wirelink_kernel_asio_udp wirelink_stream_cobs_crc16)
    add_test(NAME kernel_packet_udp COMMAND kernel_packet_udp_test)
    set_tests_properties(kernel_packet_udp PROPERTIES TIMEOUT 10)
  endif()
endif()
